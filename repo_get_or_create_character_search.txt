package com.yourname.idlerpg

import io.github.jan.supabase.postgrest.postgrest
import kotlinx.serialization.json.buildJsonObject
import kotlinx.serialization.json.put

// Bridges a local Room-database character (local_character_id) to its
// matching Supabase characters row (a separate, real uuid). Every
// chat/guild RPC call needs the Supabase uuid, not the local id -
// they are different ID schemes entirely.
object SupabaseCharacterRepository {

    private val client get() = SupabaseClientProvider.client

    suspend fun getOrCreateCharacter(localCharacterId: String, displayName: String): Result<String> = runCatching {
        client.postgrest.rpc("get_or_create_character", buildJsonObject {
            put("p_local_character_id", localCharacterId)
            put("p_display_name", displayName)
        }).decodeAs<String>()
    }
}

--- SQL FILE FOUND: ./fix_send_message_blocks_check.sql ---
-- fix_send_message_blocks_check.sql
-- Root cause (confirmed via live test): blocking a character hid their DM
-- from view (blockCharacter() only affects client-side fetch filtering) but
-- never stopped them from sending - send_chat_message checks mutes and
-- channel participation, but never touches the blocks table at all.
--
-- Fix: for DM channels, look up the other participant in the channel and
-- block the send if either side has blocked the other.

CREATE OR REPLACE FUNCTION public.send_chat_message(p_character_id uuid, p_channel_id uuid, p_content text)
 RETURNS chat_messages
 LANGUAGE plpgsql
 SECURITY DEFINER
AS $function$
declare
    v_message public.chat_messages;
    v_channel_type public.channel_type;
    v_is_participant boolean;
    v_mute_expires timestamptz;
    v_other_character_id uuid;
begin
    if not public.character_owned_by_caller(p_character_id) then
        raise exception 'Character does not belong to caller.';
    end if;

    select expires_at into v_mute_expires
        from public.mutes
        where character_id = p_character_id
          and (expires_at is null or expires_at > now())
        limit 1;
    if found then
        raise exception 'You are muted%',
            case when v_mute_expires is not null then ' until ' || v_mute_expires else '' end;
    end if;

    if trim(p_content) = '' then
        raise exception 'Message cannot be empty.';
    end if;

    select type into v_channel_type from public.chat_channels where id = p_channel_id;
    if v_channel_type is null then
        raise exception 'Channel not found.';
    end if;

    if v_channel_type in ('GUILD', 'DM') then
        select exists (
            select 1 from public.chat_channel_participants
            where channel_id = p_channel_id and character_id = p_character_id
        ) into v_is_participant;
        if not v_is_participant then
            raise exception 'Not a participant of this channel.';
        end if;
    end if;

    -- NEW: for DMs, block the send if either side has blocked the other.
    if v_channel_type = 'DM' then
        select character_id into v_other_character_id
        from public.chat_channel_participants
        where channel_id = p_channel_id and character_id != p_character_id
        limit 1;

        if v_other_character_id is not null and exists (
            select 1 from public.blocks
            where (blocker_character_id = p_character_id and blocked_character_id = v_other_character_id)
               or (blocker_character_id = v_other_character_id and blocked_character_id = p_character_id)
        ) then
            raise exception 'You cannot message this character.';
        end if;
    end if;

    insert into public.chat_messages (id, channel_id, sender_character_id, content, sent_at, type)
        values (gen_random_uuid(), p_channel_id, p_character_id, p_content, now(), 'NORMAL')
        returning * into v_message;

    return v_message;
end;
$function$
;

-- fix_blocks_insert_policy.sql
-- Root cause (confirmed via live test + pg_policies check): blocks only ever
-- had a SELECT policy ("blocks visible to the blocker"). No INSERT policy
-- existed, so blockCharacter()'s insert was silently RLS-denied every time -
-- wrapped in runCatching client-side, so it just looked like "nothing
-- happened" with no visible error.

create policy "blocks insertable by the blocker"
on public.blocks
for insert
with check (
  exists (
    select 1 from public.characters ch
    where ch.id = blocks.blocker_character_id and ch.profile_id = auth.uid()
  )
);

-- fix_guild_chat_participants.sql
-- Root cause (confirmed via live "Not a participant of this channel" error,
-- code P0001, from send_chat_message): get_or_create_guild_channel creates
-- the GUILD chat_channels row, but nothing ever inserts a matching row into

-- 1. create_guild: add founder as a participant of the new guild channel.
CREATE OR REPLACE FUNCTION public.create_guild(p_name text, p_founder_character_id uuid)
 RETURNS guilds
 LANGUAGE plpgsql
 SECURITY DEFINER
AS $function$
declare
    v_profile_id uuid := auth.uid();
    v_guild public.guilds;
    v_channel_id uuid;
begin
    if v_profile_id is null then
        raise exception 'Not authenticated';
    end if;

    if not exists (
        select 1 from public.characters
        where id = p_founder_character_id and profile_id = v_profile_id
    ) then
        raise exception 'Character does not belong to caller';
    end if;

    if length(p_name) < 3 or length(p_name) > 24 then
        raise exception 'Guild name must be between 3 and 24 characters';
    end if;

    insert into public.guilds (name, leader_character_id)
        values (p_name, p_founder_character_id)
        returning * into v_guild;

    insert into public.guild_members (guild_id, character_id, role)
        values (v_guild.id, p_founder_character_id, 'LEADER');

    v_channel_id := public.get_or_create_guild_channel(v_guild.id);

    insert into public.chat_channel_participants (channel_id, character_id)
        values (v_channel_id, p_founder_character_id)
        on conflict do nothing;

    update public.profiles
        set free_guild_creation_used = true
        where id = v_profile_id and free_guild_creation_used = false;

    return v_guild;
end;
$function$
;

-- 2. join_guild: add the joining character as a participant of the guild channel.
CREATE OR REPLACE FUNCTION public.join_guild(p_guild_id uuid, p_character_id uuid)
 RETURNS void
 LANGUAGE plpgsql
 SECURITY DEFINER
AS $function$
declare
    v_profile_id uuid := auth.uid();
    v_member_count integer;
    v_channel_id uuid;
begin
    if v_profile_id is null then
        raise exception 'Not authenticated';
    end if;

    if not exists (
        select 1 from public.characters
        where id = p_character_id and profile_id = v_profile_id
    ) then
        raise exception 'Character does not belong to caller';
    end if;

    if exists (select 1 from public.guild_members where character_id = p_character_id) then
        raise exception 'Character is already in a guild';
    end if;

    if not exists (select 1 from public.guilds where id = p_guild_id) then
        raise exception 'Guild not found';
    end if;

    select count(*) into v_member_count from public.guild_members where guild_id = p_guild_id;
    if v_member_count >= 20 then
        raise exception 'Guild is at max capacity (20)';
    end if;

    insert into public.guild_members (guild_id, character_id, role)
        values (p_guild_id, p_character_id, 'MEMBER');

    v_channel_id := public.get_or_create_guild_channel(p_guild_id);

    insert into public.chat_channel_participants (channel_id, character_id)
        values (v_channel_id, p_character_id)
        on conflict do nothing;
end;
$function$
;

-- fix_create_guild_channel.sql
-- Root cause (confirmed via pg_get_functiondef): create_guild() inserts the
-- guild row and the founder's guild_members row, then returns - it never
-- calls get_or_create_guild_channel. That's why no GUILD-type chat_channels
-- row exists for BooziBar (outstanding item #4).
--
-- Fix: add the channel creation inside create_guild itself, in the same
-- transaction, since both functions are SECURITY DEFINER and this mirrors
-- the existing server-enforced pattern used everywhere else in this project.

CREATE OR REPLACE FUNCTION public.create_guild(p_name text, p_founder_character_id uuid)
 RETURNS guilds
 LANGUAGE plpgsql
 SECURITY DEFINER
AS $function$
declare
    v_profile_id uuid := auth.uid();
    v_guild public.guilds;
begin
    if v_profile_id is null then
        raise exception 'Not authenticated';
    end if;

    if not exists (
        select 1 from public.characters
        where id = p_founder_character_id and profile_id = v_profile_id
    ) then
        raise exception 'Character does not belong to caller';
    end if;

    if length(p_name) < 3 or length(p_name) > 24 then
        raise exception 'Guild name must be between 3 and 24 characters';
    end if;

    insert into public.guilds (name, leader_character_id)
        values (p_name, p_founder_character_id)
        returning * into v_guild;

    insert into public.guild_members (guild_id, character_id, role)
        values (v_guild.id, p_founder_character_id, 'LEADER');

    -- NEW: create the guild's chat channel in the same transaction

-- Auto-create a profiles row whenever a new auth.users row appears.
-- Username is derived from the email's local part since nothing else
-- is available at signup time (the app doesn't currently pass a
-- username via signUpWith's metadata - worth adding later so this
-- can use the real chosen username instead of guessing from email).

create or replace function public.handle_new_auth_user()
returns trigger
language plpgsql
security definer
as $$
begin
    insert into public.profiles (id, username, created_at, free_guild_creation_used, is_moderator)
    values (
        new.id,
        split_part(new.email, '@', 1),
        now(),
        false,
        false
    )
    on conflict (id) do nothing;
    return new;
end;
$$;

drop trigger if exists on_auth_user_created on auth.users;
create trigger on_auth_user_created
    after insert on auth.users
    for each row execute function public.handle_new_auth_user();
