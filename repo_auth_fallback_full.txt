93:     // creation. If this fails (no network, Supabase down, etc.) local gameplay is
94:     // completely unaffected - only guild/chat multiplayer stays unavailable.
95:     private fun syncSupabaseAuth(email: String, password: String, isNewAccount: Boolean) {
96:         viewModelScope.launch {
97:             _supabaseAuthState.value = SupabaseAuthState.SIGNING_IN
98:             try {
99:                 val auth = SupabaseClientProvider.client.auth
100:                 if (isNewAccount) {
101:                     auth.signUpWith(io.github.jan.supabase.auth.providers.builtin.Email) {
102:                         this.email = email
103:                         this.password = password
104:                     }
105:                 } else {
106:                     try {
107:                         auth.signInWith(io.github.jan.supabase.auth.providers.builtin.Email) {
108:                             this.email = email
109:                             this.password = password
110:                         }
111:                     } catch (signInError: Exception) {
112:                         // Local account existed before Supabase sync was added - no
113:                         // matching Supabase account exists yet. Lazily create one now
114:                         // with the same credentials, so future logins sign in normally.
115:                         auth.signUpWith(io.github.jan.supabase.auth.providers.builtin.Email) {
116:                             this.email = email
117:                             this.password = password
118:                         }
119:                     }
120:                 }
121:                 _supabaseAuthState.value = SupabaseAuthState.SIGNED_IN
122:             } catch (e: Exception) {
123:                 _supabaseAuthState.value = SupabaseAuthState.ERROR
124:                 _supabaseAuthError.value = e.message ?: e.toString()
125:             }
126:         }
127:     }
