class CombatService {
    private val recoveryDuration: Duration = Duration.ofSeconds(600)
    private val states: MutableMap<String, CharacterCombatState> = mutableMapOf()

    fun stateFor(characterId: String, maxHp: Int = 100): CharacterCombatState =
        states.getOrPut(characterId) { CharacterCombatState(currentHp = maxHp, maxHp = maxHp) }

    // trainingSkillLevel must be the level of whichever SkillType the character
    // currently has selected to train (loadout.trainingSkill) - NOT a blended
    // "character level". Each combat skill unlocks zones independently.
    fun enterZone(characterId: String, zoneId: String, trainingSkillLevel: Int): ServiceResult<CombatSession> {
        val state = stateFor(characterId)
        if (state.recovery.isRecovering()) {
            return ServiceResult.Failure("Cannot enter combat while recovering.")
        }
        val zone = ZoneCatalog.zones.find { it.id == zoneId }
            ?: return ServiceResult.Failure("Zone not found.")
        if (zone.levelRequirement > trainingSkillLevel) {
            return ServiceResult.Failure("Requires level " + zone.levelRequirement + " to enter " + zone.name + ".")
        }

        val session = CombatSession(zone = zone)
        state.session = session
        return startSearching(characterId)
    }

    fun startSearching(characterId: String, now: Instant = Instant.now()): ServiceResult<CombatSession> {
        val state = stateFor(characterId)
        val session = state.session ?: return ServiceResult.Failure("Not currently in a zone.")
        if (state.recovery.isRecovering(now)) {
            return ServiceResult.Failure("Cannot search while recovering.")
        }

        val searchSeconds = Random.nextInt(3, 11)
        session.state = SessionState.SEARCHING
        session.searchEndsAt = now.plusSeconds(searchSeconds.toLong())
        session.currentMonster = null
        return ServiceResult.Success(session)
    }

    fun pauseSearching(characterId: String): ServiceResult<CombatSession> {
        val state = stateFor(characterId)
        val session = state.session ?: return ServiceResult.Failure("Not currently in a zone.")
        if (session.state != SessionState.SEARCHING) {
            return ServiceResult.Failure("Can only pause while searching.")
        }
        session.state = SessionState.PAUSED
        return ServiceResult.Success(session)
    }

    fun resolveSearchIfReady(characterId: String, now: Instant = Instant.now()): ServiceResult<Monster> {
        val state = stateFor(characterId)
        val session = state.session ?: return ServiceResult.Failure("Not currently in a zone.")
        if (session.state != SessionState.SEARCHING) {
            return ServiceResult.Failure("Not currently searching.")
        }
        val endsAt = session.searchEndsAt ?: return ServiceResult.Failure("No active search timer.")
        if (now.isBefore(endsAt)) {
            return ServiceResult.Failure("Search still in progress.")
        }

        val monster = session.zone.monsters.random()
        session.currentMonster = monster
        session.currentMonsterHealth = monster.maxHealth
        session.state = SessionState.IN_COMBAT
        return ServiceResult.Success(monster)
    }

    fun leaveZone(characterId: String): ServiceResult<Unit> {
        val state = stateFor(characterId)
        state.session = null
        return ServiceResult.Success(Unit)
    }

    fun canChangeEquipment(characterId: String, slot: EquipmentSlot, now: Instant = Instant.now()): Boolean {
        if (slot == EquipmentSlot.TOOL) return true

        val state = stateFor(characterId)

        if (state.recovery.isRecovering(now) && slot == EquipmentSlot.WEAPON) return false

        val session = state.session ?: return true
        return session.state != SessionState.IN_COMBAT
    }

    fun resolveAttack(
        characterId: String,
        loadout: CombatLoadout,
        attackerAccuracy: Int,
        attackerMaxHit: Int
    ): ServiceResult<AttackResult> {
        val state = stateFor(characterId)
        val session = state.session ?: return ServiceResult.Failure("Not currently in a zone.")
        if (session.state != SessionState.IN_COMBAT) {
            return ServiceResult.Failure("Not currently in combat.")
        }
        val monster = session.currentMonster ?: return ServiceResult.Failure("No active target.")

        val hitChancePercent = (attackerAccuracy.toDouble() / (attackerAccuracy + monster.defenseRating)) * 100.0
        val didHit = Random.nextDouble(0.0, 100.0) < hitChancePercent

        if (!didHit) {
            return ServiceResult.Success(
                AttackResult(didHit = false, damageDealt = 0, monsterHpRemaining = session.currentMonsterHealth, monsterDefeated = false)
            )
        }

        val damage = Random.nextInt(0, attackerMaxHit + 1)
        session.currentMonsterHealth = (session.currentMonsterHealth - damage).coerceAtLeast(0)
        val defeated = session.currentMonsterHealth <= 0

        if (defeated) {
            rollLoot(state, monster)
            session.currentMonster = null
            startSearching(characterId)
        }

        return ServiceResult.Success(
            AttackResult(didHit = true, damageDealt = damage, monsterHpRemaining = session.currentMonsterHealth, monsterDefeated = defeated)
        )
    }

    private fun rollLoot(state: CharacterCombatState, monster: Monster) {
        (monster.mainLootTable + monster.rareLootTable).forEach { entry ->
            if (Random.nextDouble(0.0, 100.0) < entry.dropChancePercent) {
                val qty = Random.nextInt(entry.minQty, entry.maxQty + 1)
                state.lootPool.add(entry.copy(minQty = qty, maxQty = qty))
            }
        }
    }

    fun collectLoot(characterId: String): ServiceResult<List<LootEntry>> {
        val state = stateFor(characterId)
        val collected = state.lootPool.toList()
        state.lootPool.clear()
        return ServiceResult.Success(collected)
    }

    fun applyDamageToCharacter(characterId: String, damage: Int, now: Instant = Instant.now()): ServiceResult<CharacterCombatState> {
        val state = stateFor(characterId)
        state.currentHp = (state.currentHp - damage).coerceAtLeast(0)

        if (state.currentHp <= 0) {
            state.recovery.recoveringUntil = now.plus(recoveryDuration)
            state.session = null
        }
        return ServiceResult.Success(state)
    }

    fun healCharacter(characterId: String, amount: Int): ServiceResult<CharacterCombatState> {
        val state = stateFor(characterId)
        state.currentHp = (state.currentHp + amount).coerceAtMost(state.maxHp)
        return ServiceResult.Success(state)
    }

    fun resolveRecoveryIfReady(characterId: String, now: Instant = Instant.now()): ServiceResult<CharacterCombatState> {
        val state = stateFor(characterId)
        val until = state.recovery.recoveringUntil
            ?: return ServiceResult.Failure("Character is not recovering.")
        if (now.isBefore(until)) {
            return ServiceResult.Failure("Recovery still in progress.")
        }
        state.currentHp = (state.maxHp * 0.5).toInt()
        state.recovery.recoveringUntil = null
        return ServiceResult.Success(state)
    }
}

@Composable
private fun LiveCountdownText(endsAt: Instant) {
    var remaining by remember(endsAt) { mutableStateOf(Duration.between(Instant.now(), endsAt)) }

    LaunchedEffect(endsAt) {
        while (true) {
            remaining = Duration.between(Instant.now(), endsAt)
            if (remaining.isNegative || remaining.isZero) break
            delay(1000)
        }
    }

    val text = if (remaining.isNegative || remaining.isZero) {
        "Encounter starting..."
    } else {
        remaining.seconds.toString() + "s remaining"
    }

    Text(text = text, style = MaterialTheme.typography.bodyMedium)
}

object OfflineEfficiencyCurve {
    private val fullRateWindow: Duration = Duration.ofHours(12)
    private val reducedRateWindow: Duration = Duration.ofHours(72)
    private const val FULL_RATE_PERCENT = 100.0
    private const val REDUCED_RATE_PERCENT = 50.0
    private const val FLOOR_RATE_PERCENT = 25.0

    fun efficiencyAt(elapsed: Duration): Double {
        return when {
            elapsed <= fullRateWindow -> FULL_RATE_PERCENT
            elapsed <= reducedRateWindow -> REDUCED_RATE_PERCENT
            else -> FLOOR_RATE_PERCENT
        }
    }

    fun effectiveElapsed(elapsed: Duration): Duration {
        if (elapsed <= fullRateWindow) return elapsed

        val fullMillis = fullRateWindow.toMillis()
        val reducedEndMillis = minOf(elapsed.toMillis(), reducedRateWindow.toMillis())
        val reducedMillis = (reducedEndMillis - fullMillis).coerceAtLeast(0)
        val floorMillis = (elapsed.toMillis() - reducedRateWindow.toMillis()).coerceAtLeast(0)

        val effectiveMillis =
            (fullMillis * (FULL_RATE_PERCENT / 100.0)) +
            (reducedMillis * (REDUCED_RATE_PERCENT / 100.0)) +
            (floorMillis * (FLOOR_RATE_PERCENT / 100.0))

        return Duration.ofMillis(effectiveMillis.toLong())
    }
}

    val maxPlausibleElapsed: Duration = Duration.ofDays(30)

    fun validate(characterId: String, claimedElapsed: Duration): Pair<Duration, OfflineFlagRecord?> {
        if (claimedElapsed <= maxPlausibleElapsed) return claimedElapsed to null

        val flag = OfflineFlagRecord(
            characterId = characterId,
            claimedElapsed = claimedElapsed,
            cappedElapsed = maxPlausibleElapsed
        )
        return maxPlausibleElapsed to flag
    }
}

class OfflineIdleService(private val materialCheck: MaterialAvailabilityCheck) {

    private val activeQueues: MutableMap<String, ActiveQueue> = mutableMapOf()
    private val lastSeenAt: MutableMap<String, Instant> = mutableMapOf()
    private val flaggedAccounts: MutableList<OfflineFlagRecord> = mutableListOf()

    fun startSkillingQueue(characterId: String, skillId: String, itemId: String, secondsPerAction: Long): ServiceResult<ActiveQueue> {
        if (activeQueues.containsKey(characterId)) {
            return ServiceResult.Failure("Character already has an active queue. Stop it before starting a new one.")
        }
        val queue = ActiveQueue.SkillingQueue(characterId, skillId, itemId, secondsPerAction)
        activeQueues[characterId] = queue
        return ServiceResult.Success(queue)
    }

    fun startCraftingQueue(
        characterId: String,
        itemId: String,
        secondsPerAction: Long,
        targetQuantity: Int?
    ): ServiceResult<ActiveQueue> {
        if (activeQueues.containsKey(characterId)) {
            return ServiceResult.Failure("Character already has an active queue. Stop it before starting a new one.")
        }
        if (targetQuantity != null && targetQuantity <= 0) {
            return ServiceResult.Failure("Target quantity must be positive if specified.")
        }
        val queue = ActiveQueue.CraftingQueue(characterId, itemId, secondsPerAction, targetQuantity)
        activeQueues[characterId] = queue
        return ServiceResult.Success(queue)
    }

    fun optIntoOfflineCombat(characterId: String, zoneId: String): ServiceResult<ActiveQueue> {
        if (activeQueues.containsKey(characterId)) {
            return ServiceResult.Failure("Character already has an active queue. Stop it before opting into offline combat.")
        }
        val queue = ActiveQueue.OfflineCombatQueue(characterId, zoneId)
        activeQueues[characterId] = queue
        return ServiceResult.Success(queue)
    }

    fun stopQueue(characterId: String): ServiceResult<Unit> {
        if (!activeQueues.containsKey(characterId)) return ServiceResult.Failure("No active queue to stop.")
        activeQueues.remove(characterId)
        return ServiceResult.Success(Unit)
    }

    fun activeQueueFor(characterId: String): ActiveQueue? = activeQueues[characterId]

    fun recordLastSeen(characterId: String, now: Instant = Instant.now()) {
        lastSeenAt[characterId] = now
    }

    fun resolveOfflineProgress(characterId: String, now: Instant = Instant.now()): ServiceResult<OfflineResolution> {
        val queue = activeQueues[characterId] ?: return ServiceResult.Failure("No active queue to resolve.")
        val lastSeen = lastSeenAt[characterId] ?: return ServiceResult.Failure("No last-seen timestamp recorded.")

        val claimedElapsed = Duration.between(lastSeen, now)
        if (claimedElapsed.isNegative) return ServiceResult.Failure("Invalid elapsed time (negative duration).")

        val (cappedElapsed, flag) = OfflineSanityCheck.validate(characterId, claimedElapsed)
        if (flag != null) flaggedAccounts.add(flag)

        val combatPenaltyMultiplier = if (queue.type == QueueType.OFFLINE_COMBAT) 0.8 else 1.0
        val effectiveElapsed = OfflineEfficiencyCurve.effectiveElapsed(cappedElapsed)
        val penalizedEffectiveMillis = (effectiveElapsed.toMillis() * combatPenaltyMultiplier).toLong()
        val finalEffectiveElapsed = Duration.ofMillis(penalizedEffectiveMillis)

        var actionsCompleted = 0
        var queueExhausted = false

        when (queue) {
            is ActiveQueue.SkillingQueue -> {
                actionsCompleted = (finalEffectiveElapsed.seconds / queue.secondsPerAction).toInt()
            }
            is ActiveQueue.CraftingQueue -> {
                var possibleActions = (finalEffectiveElapsed.seconds / queue.secondsPerAction).toInt()
                if (queue.targetQuantity != null) {
                    val remaining = queue.targetQuantity - queue.quantityCompleted
                    possibleActions = possibleActions.coerceAtMost(remaining)
                }
                var completed = 0
                for (i in 0 until possibleActions) {
                    if (!materialCheck.hasMaterialsFor(characterId, queue.itemId)) {
                        queueExhausted = true
                        break
                    }
                    completed++
                }
                queue.quantityCompleted += completed
                actionsCompleted = completed
                if (queue.isTargetReached()) queueExhausted = true
            }
            is ActiveQueue.OfflineCombatQueue -> {
                actionsCompleted = 0
            }
        }

        if (queueExhausted) {
            activeQueues.remove(characterId)
        }

        return ServiceResult.Success(
            OfflineResolution(
                characterId = characterId,
                queueType = queue.type,
                claimedElapsed = claimedElapsed,
                cappedElapsed = cappedElapsed,
                effectiveElapsed = finalEffectiveElapsed,
                actionsCompleted = actionsCompleted,
                wasFlagged = flag != null,
                queueExhausted = queueExhausted
            )
        )
    }

    fun estimatedTimeUntilExhausted(characterId: String): Duration? {
        val queue = activeQueues[characterId] as? ActiveQueue.CraftingQueue ?: return null
        val target = queue.targetQuantity ?: return null
        val remaining = target - queue.quantityCompleted
        if (remaining <= 0) return Duration.ZERO
        return Duration.ofSeconds(remaining * queue.secondsPerAction)
    }

    fun allFlaggedAccounts(): List<OfflineFlagRecord> = flaggedAccounts.toList()

    fun exportSkillingQueues(): List<SkillingQueueExport> =
        activeQueues.values.filterIsInstance<ActiveQueue.SkillingQueue>().map {
            SkillingQueueExport(it.characterId, it.skillId, it.itemId, it.secondsPerAction, it.startedAt.toString())
        }

    fun exportCraftingQueues(): List<CraftingQueueExport> =
        activeQueues.values.filterIsInstance<ActiveQueue.CraftingQueue>().map {
            CraftingQueueExport(
                it.characterId,
                it.itemId,
                it.secondsPerAction,
                it.targetQuantity,
                it.quantityCompleted,
                it.startedAt.toString()
            )
        }

    fun exportLastSeen(): List<LastSeenExport> =
        lastSeenAt.map { (characterId, instant) -> LastSeenExport(characterId, instant.toString()) }

    fun restoreSkillingQueue(export: SkillingQueueExport) {
        activeQueues[export.characterId] = ActiveQueue.SkillingQueue(
            characterId = export.characterId,
            skillId = export.skillId,
            itemId = export.itemId,
            secondsPerAction = export.secondsPerAction,
            startedAt = Instant.parse(export.startedAt)
        )
    }

    fun restoreCraftingQueue(export: CraftingQueueExport) {
        val queue = ActiveQueue.CraftingQueue(
            characterId = export.characterId,
            itemId = export.itemId,
            secondsPerAction = export.secondsPerAction,
            targetQuantity = export.targetQuantity,
            startedAt = Instant.parse(export.startedAt)
        )
        queue.quantityCompleted = export.quantityCompleted
        activeQueues[export.characterId] = queue
    }

    fun restoreLastSeen(export: LastSeenExport) {
        lastSeenAt[export.characterId] = Instant.parse(export.lastSeenAt)
    }

            var count = 0
