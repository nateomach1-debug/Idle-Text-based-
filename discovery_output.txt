1111:     private data class CraftBatchResult(val successfulCrafts: Int, val itemsProduced: Int, val xpGranted: Long)

1113:     private fun craftItems(characterId: String, itemId: String, attempts: Int): CraftBatchResult {
1114:         val recipe = RecipeCatalog.forOutput(itemId) ?: return CraftBatchResult(0, 0, 0)
1115:         val skillType = SkillType.valueOf(recipe.skillRequired)
1116:         val book = progressionService.skillBookFor(characterId)
1117:         val skill = book.all().find { it.type == skillType }
1118:         val playerLevel = if (skill != null) LevelCurve.levelForXp(skill.xp) else 1
1119:         val inventory = inventoryService.inventoryFor(characterId)
1120:         val xpPerAttempt = CraftingTierCurve.xpFor(recipe.levelRequired)
1121: 
1122:         var successes = 0
1123:         var totalProduced = 0
1124:         var xpGranted = 0L
1125:         for (i in 0 until attempts) {
1126:             val result = craftingService.craftOnce(inventory, itemId, playerLevel)
1127:             if (result is ServiceResult.Success) {
1128:                 xpGranted += xpPerAttempt
1129:                 if (result.value.succeeded) {
1130:                     successes += 1
1131:                     totalProduced += result.value.quantityProduced
1132:                 }
1133:             } else {
1134:                 break
1135:             }
1136:         }
1137:         if (xpGranted > 0) {
1138:             progressionService.grantXp(characterId, skillType, xpGranted)
1139:         }
1140:         return CraftBatchResult(successes, totalProduced, xpGranted)
1141:     }

1144:         val characterId = _currentCharacterId.value ?: return

1168:     private fun tickCrafting() {
1169:         val characterId = _currentCharacterId.value ?: return
1170:         val queue = offlineIdleService.activeQueueFor(characterId) as? ActiveQueue.CraftingQueue ?: return
1171: 
1172:         craftActionProgressSeconds += 1
1173:         if (craftActionProgressSeconds < queue.secondsPerAction) return
1174: 
1175:         val attempts = (craftActionProgressSeconds / queue.secondsPerAction).toInt()
1176:         craftActionProgressSeconds %= queue.secondsPerAction
1177: 
1178:         val remainingToTarget = queue.targetQuantity?.let { it - queue.quantityCompleted }
1179:         val cappedAttempts = if (remainingToTarget != null) minOf(attempts, remainingToTarget) else attempts
1180:         if (cappedAttempts <= 0) return
1181: 
1182:         val batch = craftItems(characterId, queue.itemId, cappedAttempts)
1183:         queue.quantityCompleted += batch.successfulCrafts
1184:         refreshSkills()
1185:         refreshInventory()
1186: 
1187:         if (queue.isTargetReached()) {
1188:             offlineIdleService.stopQueue(characterId)
1189:             refreshActiveQueue()
1190:         }
1191:     }

1199:             val xpAtNextLevel = LevelCurve.xpRequiredForLevel(level + 1)

1204:                 level = level,
1205:                 currentXp = skill.xp - xpAtLevelStart,
1206:                 xpToNextLevel = xpAtNextLevel - skill.xp

1208:         }
