155: /**
156:  * Skilling/crafting loops repeat every `secondsPerAction` and don't have a
157:  * fixed overall end -- so this ticks down the time remaining in the
158:  * *current* cycle, computed live from startedAt + secondsPerAction against
159:  * wall-clock time, and rolls back to the full duration once the next cycle
160:  * begins. Ticks every 100ms so short cycles still show a smooth countdown.
161:  */
162: @Composable
163: private fun CycleCountdown(queue: ActiveQueueInfo) {
164:     fun remainingSecondsNow(): Double {
165:         val elapsedMs = Duration.between(queue.startedAt, Instant.now()).toMillis().coerceAtLeast(0)
166:         val cycleMs = queue.secondsPerAction * 1000
167:         if (cycleMs <= 0) return 0.0
168:         val remainingMs = cycleMs - (elapsedMs % cycleMs)
169:         return remainingMs / 1000.0
170:     }
171: 
172:     var remainingSeconds by remember(queue.startedAt, queue.secondsPerAction) {
173:         mutableStateOf(remainingSecondsNow())
174:     }
175: 
176:     LaunchedEffect(queue.startedAt, queue.secondsPerAction) {
177:         while (true) {
178:             remainingSeconds = remainingSecondsNow()
179:             delay(100)
180:         }
181:     }
182: 
183:     val displayText = if (remainingSeconds >= 60.0) {
184:         val totalSeconds = remainingSeconds.toLong()
185:         val minutes = totalSeconds / 60
186:         val seconds = totalSeconds % 60
187:         String.format("%d:%02d remaining", minutes, seconds)
188:     } else {
189:         String.format("%.2fs left", remainingSeconds)
190:     }
191: 
192:     Text(text = displayText, style = MaterialTheme.typography.bodyMedium)
193: }
