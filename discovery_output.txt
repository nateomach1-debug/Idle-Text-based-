240:         characterId: String,
241:         loadout: CombatLoadout,

243:         attackerMaxHit: Int
244:     ): ServiceResult<AttackResult> {
245:         val state = stateFor(characterId)
246:         val session = state.session ?: return ServiceResult.Failure("Not currently in a zone.")
247:         if (session.state != SessionState.IN_COMBAT) {
248:             return ServiceResult.Failure("Not currently in combat.")
249:         }
250:         val monster = session.currentMonster ?: return ServiceResult.Failure("No active target.")
251: 
252:         val hitChancePercent = (attackerAccuracy.toDouble() / (attackerAccuracy + monster.defenseRating)) * 100.0
253:         val didHit = Random.nextDouble(0.0, 100.0) < hitChancePercent
254: 
255:         if (!didHit) {
256:             return ServiceResult.Success(
257:                 AttackResult(didHit = false, damageDealt = 0, monsterHpRemaining = session.currentMonsterHealth, monsterDefeated = false)
258:             )
259:         }
260: 
261:         val damage = Random.nextInt(0, attackerMaxHit + 1)
262:         session.currentMonsterHealth = (session.currentMonsterHealth - damage).coerceAtLeast(0)
263:         val defeated = session.currentMonsterHealth <= 0
264: 
265:         if (defeated) {
266:             rollLoot(state, monster)
267:             session.currentMonster = null
268:             startSearching(characterId)
269:         }
270: 
271:         return ServiceResult.Success(
272:             AttackResult(didHit = true, damageDealt = damage, monsterHpRemaining = session.currentMonsterHealth, monsterDefeated = defeated)
273:         )
274:     }

61:     private val fullRateWindow: Duration = Duration.ofHours(12)
62:     private val reducedRateWindow: Duration = Duration.ofHours(72)
63:     private const val FULL_RATE_PERCENT = 100.0
64:     private const val REDUCED_RATE_PERCENT = 50.0
65:     private const val FLOOR_RATE_PERCENT = 25.0
66: 
67:     fun efficiencyAt(elapsed: Duration): Double {
68:         return when {
69:             elapsed <= fullRateWindow -> FULL_RATE_PERCENT
70:             elapsed <= reducedRateWindow -> REDUCED_RATE_PERCENT
71:             else -> FLOOR_RATE_PERCENT
72:         }
73:     }
74: 
75:     fun effectiveElapsed(elapsed: Duration): Duration {
76:         if (elapsed <= fullRateWindow) return elapsed
77: 
78:         val fullMillis = fullRateWindow.toMillis()
79:         val reducedEndMillis = minOf(elapsed.toMillis(), reducedRateWindow.toMillis())
80:         val reducedMillis = (reducedEndMillis - fullMillis).coerceAtLeast(0)
81:         val floorMillis = (elapsed.toMillis() - reducedRateWindow.toMillis()).coerceAtLeast(0)
82: 
83:         val effectiveMillis =
84:             (fullMillis * (FULL_RATE_PERCENT / 100.0)) +
85:             (reducedMillis * (REDUCED_RATE_PERCENT / 100.0)) +
86:             (floorMillis * (FLOOR_RATE_PERCENT / 100.0))
87: 
88:         return Duration.ofMillis(effectiveMillis.toLong())
89:     }
90: }
