object SupabaseGuildRepository {

    private val client get() = SupabaseClientProvider.client

    // Finds which guild (if any) a character belongs to, and their role in it.
    suspend fun findMembership(characterId: String): Pair<String, GuildMemberRole>? {
        return try {
            val row = client.postgrest.from("guild_members")
                .select { filter { eq("character_id", characterId) } }
                .decodeSingleOrNull<GuildMemberRow>() ?: return null
            row.guildId to GuildMemberRole.valueOf(row.role)
        } catch (e: Exception) {
            null
        }
    }

    suspend fun fetchGuildHome(characterId: String): GuildHomeState? {
        try {
            val (guildId, viewerRole) = findMembership(characterId) ?: return null

            val guild = client.postgrest.from("guilds")
                .select { filter { eq("id", guildId) } }
                .decodeSingleOrNull<GuildRow>() ?: return null

            val memberRows = client.postgrest.from("guild_members")
                .select { filter { eq("guild_id", guildId) } }
                .decodeList<GuildMemberRow>()

            val characterIds = memberRows.map { it.characterId }
            val names = if (characterIds.isEmpty()) emptyList() else client.postgrest.from("characters")
                .select { filter { isIn("id", characterIds) } }
                .decodeList<CharacterNameRow>()
            val nameById = names.associateBy({ it.id }, { it.displayName })

            val members = memberRows.map {
                GuildMemberSummary(
                    characterId = it.characterId,
                    displayName = nameById[it.characterId] ?: "Unknown",
                    role = GuildMemberRole.valueOf(it.role)
                )
            }

            val activeEvent = client.postgrest.from("guild_events")
                .select { filter { eq("guild_id", guildId) } }
                .decodeList<GuildEventRow>()
                .filter { Instant.parse(it.endsAt).isAfter(Instant.now()) }
                .maxByOrNull { it.startsAt }
                ?.let { GuildEventSummary(name = it.eventType, progressCurrent = 0, progressTarget = 0) }
            // Note: live progress (items contributed vs scaled target) isn't tracked in
            // guild_events yet - Phase 2 adds that. Shows the event name/existence for now.

            val openVoteRow = client.postgrest.from("guild_leader_votes")
                .select { filter { eq("guild_id", guildId); eq("status", "OPEN") } }
                .decodeList<GuildVoteRow>()
                .firstOrNull()

            val activeVote = openVoteRow?.let { vote ->
                val ballots = client.postgrest.from("guild_leader_vote_ballots")
                    .select { filter { eq("vote_id", vote.id) } }
                    .decodeList<VoteBallotRow>()
                val removeVotes = ballots.count { !it.supports }
                val keepVotes = ballots.count { it.supports }
                val targetName = nameById[vote.targetCharacterId] ?: "Unknown"
                val closesAt = Instant.parse(vote.createdAt).plus(Duration.ofHours(24))
                val hoursRemaining = Duration.between(Instant.now(), closesAt).toHours().coerceAtLeast(0)
                GuildVoteSummary(
                    targetLeaderName = targetName,
                    hoursRemaining = hoursRemaining.toInt(),
                    removeVotes = removeVotes,
                    keepVotes = keepVotes
                )
            }

            return GuildHomeState(
                guildName = guild.name,
                activeEvent = activeEvent,
                activeVote = activeVote,
                tier = guild.tier,
                activeBonusPercent = 0.0,  // Phase 2 adds active bonus tracking
                members = members,
                quests = emptyList(),  // no backend concept for this yet - not fabricating fake data
                viewerRole = viewerRole
            )
        } catch (e: Exception) {
            return null
        }
    }

    suspend fun fetchLeaderboard(characterId: String): List<GuildLeaderboardEntry> {
        // Placeholder until Phase 2 adds contribution tracking columns - returns
        // members with a value of 0 rather than fabricating numbers.
        val (guildId, _) = findMembership(characterId) ?: return emptyList()
        return try {
            val memberRows = client.postgrest.from("guild_members")
                .select { filter { eq("guild_id", guildId) } }
                .decodeList<GuildMemberRow>()
            val characterIds = memberRows.map { it.characterId }
            val names = if (characterIds.isEmpty()) emptyList() else client.postgrest.from("characters")
                .select { filter { isIn("id", characterIds) } }
                .decodeList<CharacterNameRow>()
            names.map { GuildLeaderboardEntry(displayName = it.displayName, value = 0L) }
        } catch (e: Exception) {
            emptyList()
        }
    }

            put("p_target_character_id", targetCharacterId)
        })

        client.postgrest.rpc("disband_guild", buildJsonObject {
            put("p_guild_id", guildId)
            put("p_acting_character_id", actingCharacterId)
        })
    }

    // Lists every guild so a character with no guild yet can browse and self-join.
    suspend fun listGuilds(): List<GuildBrowseEntry> {
        return try {
            val guildRows = client.postgrest.from("guilds").select().decodeList<GuildRow>()
            val memberRows = client.postgrest.from("guild_members").select().decodeList<GuildMemberRow>()
            val countByGuild = memberRows.groupingBy { it.guildId }.eachCount()
            guildRows.map { GuildBrowseEntry(it.id, it.name, it.tier, countByGuild[it.id] ?: 0) }
        } catch (e: Exception) {
            emptyList()
        }
    }

    suspend fun joinGuild(guildId: String, characterId: String): Result<Unit> = runCatching {
        client.postgrest.rpc("join_guild", buildJsonObject {
            put("p_guild_id", guildId)
            put("p_character_id", characterId)
        })
    }

class GameStateRepository(private val context: Context) {
    private val saveFile: File get() = File(context.filesDir, "game_state.json")

    fun save(
        accounts: List<AccountExport>,
        characters: List<CharacterExport>,
        skills: List<SkillExport>,
        skillingQueues: List<SkillingQueueExport>,
        craftingQueues: List<CraftingQueueExport>,
        lastSeen: List<LastSeenExport>,
        inventoryItems: List<InventorySlotExport>,
        equippedSlots: List<EquippedSlotExport>,
        loadoutSettings: List<LoadoutSettingsExport>,
        marketAccounts: List<PlayerEconomyAccountExport>,
        marketOrders: List<MarketOrderExport>,
        marketPriceHistory: List<PriceHistoryExport>
    ) {
        val root = JSONObject()

        val accountsArray = JSONArray()
        accounts.forEach { acc ->
            val obj = JSONObject()
            obj.put("id", acc.id)
            obj.put("email", acc.email)
            obj.put("username", acc.username)
            obj.put("passwordHash", acc.passwordHash)
            obj.put("passwordSalt", acc.passwordSalt)
            obj.put("createdAt", acc.createdAt)
            obj.put("lastLoginAt", acc.lastLoginAt)
            obj.put("ageBracket", acc.ageBracket)
            obj.put("region", acc.region)
            obj.put("isBanned", acc.isBanned)
            obj.put("banReason", acc.banReason)
            obj.put("mutedUntil", acc.mutedUntil)
            obj.put("subscriptionTier", acc.subscriptionTier)
            obj.put("premiumCurrency", acc.premiumCurrency)
            accountsArray.put(obj)
        }
        root.put("accounts", accountsArray)

        val charactersArray = JSONArray()
        characters.forEach { char ->
            val obj = JSONObject()
            obj.put("id", char.id)
            obj.put("accountId", char.accountId)
            obj.put("displayName", char.displayName)
            obj.put("avatarId", char.avatarId)
            obj.put("createdAt", char.createdAt)
            obj.put("displayNameLastChangedAt", char.displayNameLastChangedAt)
            obj.put("isMainCharacter", char.isMainCharacter)
            charactersArray.put(obj)
        }
        root.put("characters", charactersArray)

        val skillsArray = JSONArray()
        skills.forEach { skill ->
            val obj = JSONObject()
            obj.put("characterId", skill.characterId)
            obj.put("skillId", skill.skillId)
            obj.put("xp", skill.xp)
            skillsArray.put(obj)
        }
        root.put("skills", skillsArray)

        val queuesArray = JSONArray()
        skillingQueues.forEach { queue ->
            val obj = JSONObject()
            obj.put("characterId", queue.characterId)
            obj.put("skillId", queue.skillId)
            obj.put("itemId", queue.itemId)
            obj.put("secondsPerAction", queue.secondsPerAction)
            obj.put("startedAt", queue.startedAt)
            queuesArray.put(obj)
        }
        root.put("skillingQueues", queuesArray)

        val craftingQueuesArray = JSONArray()
        craftingQueues.forEach { queue ->
            val obj = JSONObject()
            obj.put("characterId", queue.characterId)
            obj.put("itemId", queue.itemId)
            obj.put("secondsPerAction", queue.secondsPerAction)
            obj.put("targetQuantity", queue.targetQuantity)
            obj.put("quantityCompleted", queue.quantityCompleted)
            obj.put("startedAt", queue.startedAt)
            craftingQueuesArray.put(obj)
        }
        root.put("craftingQueues", craftingQueuesArray)

        val lastSeenArray = JSONArray()
        lastSeen.forEach { entry ->
            val obj = JSONObject()
            obj.put("characterId", entry.characterId)
            obj.put("lastSeenAt", entry.lastSeenAt)
            lastSeenArray.put(obj)
        }
        root.put("lastSeen", lastSeenArray)

        val inventoryArray = JSONArray()
        inventoryItems.forEach { entry ->
            val obj = JSONObject()
            obj.put("characterId", entry.characterId)
            obj.put("itemId", entry.itemId)
            obj.put("quantity", entry.quantity)
            inventoryArray.put(obj)
        }
        root.put("inventoryItems", inventoryArray)

        val equippedArray = JSONArray()
        equippedSlots.forEach { entry ->
            val obj = JSONObject()
            obj.put("characterId", entry.characterId)
            obj.put("slot", entry.slot)
            obj.put("itemId", entry.itemId)
            equippedArray.put(obj)
        }
        root.put("equippedSlots", equippedArray)

        val loadoutSettingsArray = JSONArray()
        loadoutSettings.forEach { entry ->
            val obj = JSONObject()
            obj.put("characterId", entry.characterId)
            obj.put("selectedFoodId", entry.selectedFoodId ?: JSONObject.NULL)
            obj.put("selectedPotionId", entry.selectedPotionId ?: JSONObject.NULL)
            obj.put("potionTimerRemainingSeconds", entry.potionTimerRemainingSeconds)
            obj.put("autoHealThresholdPercent", entry.autoHealThresholdPercent)
            obj.put("autoFleeThresholdPercent", entry.autoFleeThresholdPercent)
            loadoutSettingsArray.put(obj)
        }
        root.put("loadoutSettings", loadoutSettingsArray)

        val marketAccountsArray = JSONArray()
        marketAccounts.forEach { acc ->
            val obj = JSONObject()
            obj.put("characterId", acc.characterId)
            obj.put("accountId", acc.accountId)
            obj.put("isMainCharacter", acc.isMainCharacter)
            obj.put("currencyBalance", acc.currencyBalance)
            val inventoryObj = JSONObject()
            acc.inventory.forEach { (itemId, qty) -> inventoryObj.put(itemId, qty) }
            obj.put("inventory", inventoryObj)
            marketAccountsArray.put(obj)

        val accounts = mutableListOf<AccountExport>()

        for (i in 0 until accountsArray.length()) {
            val obj = accountsArray.getJSONObject(i)
            accounts.add(
                AccountExport(
                    id = obj.getString("id"), email = obj.getString("email"),
                    username = obj.getString("username"), passwordHash = obj.getString("passwordHash"),
                    passwordSalt = obj.getString("passwordSalt"),
                    createdAt = obj.getString("createdAt"), lastLoginAt = obj.getString("lastLoginAt"),
                    ageBracket = obj.getString("ageBracket"), region = obj.getString("region"),
                    isBanned = obj.getBoolean("isBanned"),
